rf_model <- randomForest(
  x = data[, 2:ncol(data)],  # Exclude 'sample' and 'group' columns
  y = as.factor(data$group),
  ntree = 300,                # Increase the number of trees to enhance stability of importance measures
  mtry = floor(sqrt(ncol(data)-1)),  # mtry = sqrt(number of features)
  nodesize = 10,               # Increase minimum size of terminal nodes to enhance selection strictness
  importance = TRUE,           # Importance calculation must be enabled
  #cutoff = c(0.8, 0.2),       # Adjust classification threshold for class imbalance
  do.trace = 100               # Display progress every 100 trees
)
pdf("ECM_forest.pdf", 5.5, 5.5)
plot(rf_model)
dev.off()
# Extract variable importance (using MeanDecreaseGini)
var_imp <- importance(rf_model, type = 2, scale = TRUE) %>% 
  as.data.frame() %>%
  arrange(desc(MeanDecreaseGini))
var_imp
