results <- rfe(
  x = data[, 2:ncol(data)],  # Feature matrix (excluding the group column)
  y = data$group,            # Response variable
  sizes = seq(1, ncol(data)-1, by = 1), # Evaluate subsets of genes from 1 to 20
  rfeControl = rfeControl(
    functions = caretFuncs,
    method = "cv"),
  method = "svmRadial"
)
# Extract the optimal number of genes
optimal_size <- results$optsize
cat("Optimal number of features:", optimal_size, "\n")

# Obtain the list of selected genes
selected_genes <- predictors(results)
print(selected_genes)

write.csv(selected_genes, file = 'ECM_svm_rfe_gene.csv')
# Simple visualization of SVM-RFE results
pdf("ECM_svm_rfe_error.pdf", 5.5, 5.5)
